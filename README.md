        const roi = totalCost > 0 ? (netProfit / totalCost) * 100 : 0;
        const margin = sale > 0 ? (netProfit / sale) * 100 : 0;

        const feeDecimal = feePercent / 100;
        const breakEven = feeDecimal < 1 ? (totalCost + ship) / (1 - feeDecimal) : 0;
        
        // Max buy price (targeting 30% ROI)
        const netRevenue = sale - platformFee - ship;
        const maxBuy = netRevenue / 1.30;

        // Render Results
        profitEl.textContent = (netProfit >= 0 ? '€' : '-€') + Math.abs(netProfit).toFixed(2);
        profitEl.className = netProfit >= 0 ? 'text-2xl font-black text-lime mt-1 block font-mono' : 'text-2xl font-black text-rose-400 mt-1 block font-mono';

        roiEl.textContent = roi.toFixed(1) + '%';
        marginEl.textContent = margin.toFixed(1) + '%';
        breakevenEl.textContent = '€' + breakEven.toFixed(2);
        maxbuyEl.textContent = '€' + (maxBuy > 0 ? maxBuy.toFixed(2) : '0.00');

        if (roi >= 40 && netProfit >= 15) {
            badgeEl.textContent = 'VERDICT: BUY';
            badgeEl.className = 'px-3 py-1 rounded text-xs font-extrabold bg-lime/20 text-lime border border-lime/40';
        } else if (roi >= 15 && netProfit > 0) {
            badgeEl.textContent = 'VERDICT: MAYBE';
            badgeEl.className = 'px-3 py-1 rounded text-xs font-extrabold bg-amber-500/20 text-amber-400 border border-amber-500/40';
        } else {
            badgeEl.textContent = 'VERDICT: PASS';
            badgeEl.className = 'px-3 py-1 rounded text-xs font-extrabold bg-rose-500/20 text-rose-400 border border-rose-500/40';
        }
    }
